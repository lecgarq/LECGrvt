    ```csharp
    using Autodesk.Revit.UI.Events;
    using LECG.Services.Logging;

    namespace LECG.Core;

    public static class DialogWhitelist
    {
        private const string Scope = "DialogWhitelist";

        // SOURCE: .planning/phases/06-cross-cutting-foundation/06-DIALOG-DISCOVERY.md
        // Reviewer: extend only via deliberate code change.
        private static readonly IReadOnlyDictionary<string, int> Entries = new Dictionary<string, int>(StringComparer.Ordinal)
        {
            // Populate from 06-DIALOG-DISCOVERY.md.
            // If discovery captured nothing (blocked checkpoint), seed with the LOW-confidence entries
            // from RESEARCH.md "Proposed whitelist entries" and flag each with a // TODO: confirm at runtime
            // comment so the next time Purge / ConvertFamily run, the entries can be validated.
            // Example shape:
            // { "TaskDialog_Overwrite", 1 },           // ConvertFamily: accept overwrite
            // { "TaskDialog_DuplicateFamily", 1 },     // ConvertFamily: accept duplicate
            // { "TaskDialog_ExtrusionTooThin", 2 },    // Purge: cancel; preserve geometry
        };

        public static void Apply(IDialogOverride sink, ILogger logger)
        {
            if (sink is null) throw new ArgumentNullException(nameof(sink));
            if (logger is null) throw new ArgumentNullException(nameof(logger));

            var id = sink.DialogId;
            if (id is not null && Entries.TryGetValue(id, out var result))
            {
                sink.OverrideResult(result);
                logger.Log($"Auto-handled whitelisted dialog '{id}' with result {result}", scope: Scope);
                return;
            }

            logger.LogWarning($"Dialog '{id ?? "(null)"}' not in whitelist — reaching user", scope: Scope);
        }

        public static void Apply(DialogBoxShowingEventArgs e, ILogger logger)
        {
            if (e is null) throw new ArgumentNullException(nameof(e));
            Apply(new DialogOverrideAdapter(e), logger);
        }

        private sealed class DialogOverrideAdapter : IDialogOverride
        {
            private readonly DialogBoxShowingEventArgs _e;
            public DialogOverrideAdapter(DialogBoxShowingEventArgs e) { _e = e; }
            public string? DialogId => _e.DialogId;
            public void OverrideResult(int result) => _e.OverrideResult(result);
        }
    }
    ```
