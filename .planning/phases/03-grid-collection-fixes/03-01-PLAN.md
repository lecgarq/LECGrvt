    ```csharp
    public static class ElementLabelService
    {
        public static (string name, string category) GetLabelsFromRaw(
            string rawName, string rawCategory, string clrTypeName, long id)
        {
            string name = string.IsNullOrWhiteSpace(rawName)
                ? $"<{clrTypeName} {id}>"
                : rawName.Trim();
            string category = string.IsNullOrWhiteSpace(rawCategory)
                ? (string.IsNullOrWhiteSpace(clrTypeName) ? "Element" : clrTypeName)
                : rawCategory.Trim();
            return (name, category);
        }

        public static (string name, string category) GetLabels(Autodesk.Revit.DB.Element element)
        {
            // 1) Resolve raw name. Fallback to BuiltInParameter.ALL_MODEL_TYPE_NAME for
            //    localized Revit per RESEARCH §Pitfall 6.
            string rawName = element.Name;
            if (string.IsNullOrWhiteSpace(rawName))
            {
                var p = element.get_Parameter(Autodesk.Revit.DB.BuiltInParameter.ALL_MODEL_TYPE_NAME);
                rawName = p?.AsString() ?? "";
            }

            // 2) Resolve raw category locale-safely.
            string rawCategory = "";
            var cat = element.Category;
            if (cat != null)
            {
                try
                {
                    var bic = (Autodesk.Revit.DB.BuiltInCategory)cat.Id.Value;
                    rawCategory = Autodesk.Revit.DB.LabelUtils.GetLabelFor(bic);
                }
                catch { rawCategory = cat.Name ?? ""; }
            }

            string clrTypeName = element.GetType().Name;
            long id = element.Id.Value;

            var labels = GetLabelsFromRaw(rawName, rawCategory, clrTypeName, id);

            // 3) Logging hook: when a fallback fired, emit LogView warning per CONTEXT.md.
            if (string.IsNullOrWhiteSpace(rawCategory) || string.IsNullOrWhiteSpace(element.Name))
            {
                try { Logger.Instance.LogWarning($"ElementLabelService: fallback for {clrTypeName} {id} → ({labels.name}, {labels.category})"); }
                catch { /* logger may be unavailable in some contexts */ }
            }

            return labels;
        }
    }
    ```

    DO NOT use `(BuiltInCategory)element.Category?.Id.Value` chain shown in RESEARCH §Pattern 2 — it's incorrect (the `?.` short-circuits to nullable long). The correct pattern is the explicit null-check above. Note RESEARCH §Pattern 2 in code-review hints rather than copying verbatim.
