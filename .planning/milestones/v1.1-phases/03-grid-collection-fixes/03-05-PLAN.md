    1. In `SearchReplaceViewModel.cs`, add an ICollectionView wrapping `PreviewItems`:

       ```csharp
       using System.ComponentModel;
       using System.Windows.Data;

       private ICollectionView? _previewView;
       public ICollectionView PreviewView
       {
           get
           {
               if (_previewView == null)
               {
                   _previewView = CollectionViewSource.GetDefaultView(PreviewItems);
                   _previewView.SortDescriptions.Add(new SortDescription(
                       nameof(ElementRowViewModel.Category),
                       ListSortDirection.Ascending));
                   _previewView.Filter = item => MatchesAllFilters((ElementRowViewModel)item);
               }
               return _previewView;
           }
       }

       // Container for per-column filters. Per-column filter UI in Task 1's note.
       private readonly Dictionary<string, Predicate<ElementRowViewModel>> _columnFilters = new();

       private bool MatchesAllFilters(ElementRowViewModel row)
       {
           // Top-of-grid dropdown
           if (!string.IsNullOrEmpty(FilterCategory) && !FilterCategory.Equals("All", StringComparison.OrdinalIgnoreCase))
           {
               if (!row.Category.Equals(FilterCategory, StringComparison.OrdinalIgnoreCase)) return false;
           }
           // Per-column predicates (AND)
           foreach (var p in _columnFilters.Values) if (!p(row)) return false;
           return true;
       }

       partial void OnFilterCategoryChanged(string? value) => _previewView?.Refresh();
       ```

       Hooking the FilterCategory setter: if the property uses `[ObservableProperty]`, use the `OnFilterCategoryChanged` partial; otherwise wire `Refresh()` in the existing setter.
